# agents_log.py - drop this import into every agent
import json, datetime, pathlib

LOG_PATH = pathlib.Path("C:/agents/agents.log")

def log_event(agent_name: str, event: str, **kwargs):
    entry = {
        "agent": agent_name,
        "event": event,  # "start" or "finish"
        "timestamp": datetime.datetime.now().isoformat(),
        **kwargs
    }
    with open(LOG_PATH, "a") as f:
        f.write(json.dumps(entry) + "\n")

# anomaly_check.py - simplified version
def check_anomaly(agent_name: str, metric: str, current_value: float,
                   history: list[float]) -> bool:
    if len(history) < 3:
        return False  # not enough history to flag anomalies
    avg = sum(history[-7:]) / len(history[-7:])
    if avg == 0:
        return current_value > 5  # any non-trivial output is anomalous
    ratio = current_value / avg
    return ratio > 2.5 or ratio < 0.2  # 2.5x over or 80% under

python
# monitor.py - runs every hour via Task Scheduler
import json
import datetime
import requests
import pathlib

LOG_PATH = pathlib.Path("C:/agents/agents.log")
TELEGRAM_TOKEN = "YOUR_BOT_TOKEN"
CHAT_ID = "YOUR_CHAT_ID"

# Expected run cadence in hours for each agent
AGENT_CADENCE = {
    "scraper": 2,
    "twitter_poster": 1,
    "briefing_agent": 24,
    "stats_server": 0.5,
}

def send_telegram(message: str):
    url = f"https://api.telegram.org/bot{TELEGRAM_TOKEN}/sendMessage"
    requests.post(url, json={"chat_id": CHAT_ID, "text": message})

def load_recent_events() -> dict:
    """Returns the most recent finish event for each agent."""
    latest = {}
    if not LOG_PATH.exists():
        return latest
    with open(LOG_PATH, "r") as f:
        for line in f:
            try:
                entry = json.loads(line.strip())
                if entry.get("event") == "finish":
                    agent = entry["agent"]
                    ts = datetime.datetime.fromisoformat(entry["timestamp"])
                    if agent not in latest or ts > latest[agent]["ts"]:
                        latest[agent] = {"ts": ts, "status": entry.get("status"),
                                          "count": entry.get("count", 0)}
            except (json.JSONDecodeError, KeyError):
                continue
    return latest

def check_agents():
    now = datetime.datetime.now()
    latest = load_recent_events()
    alerts = []

    for agent, cadence_hours in AGENT_CADENCE.items():
        if agent not in latest:
            alerts.append(f"⚠️ {agent}: no run recorded. First run expected.")
            continue
        hours_since = (now - latest[agent]["ts"]).total_seconds() / 3600
        if hours_since > cadence_hours * 1.5:
            alerts.append(
                f"⚠️ {agent}: last run {hours_since:.1f}h ago "
                f"(expected every {cadence_hours}h). "
                f"Last status: {latest[agent]['status']}"
            )
        elif latest[agent]["status"] != "success":
            alerts.append(
                f"❌ {agent}: last run FAILED at "
                f"{latest[agent]['ts'].strftime('%H:%M')}. Check logs."
            )

    if alerts:
        message = "🤖 MONITORING ALERT - " + now.strftime("%H:%M %b %d") + "\n\n"
        message += "\n".join(alerts)
        send_telegram(message)

if __name__ == "__main__":
    check_agents()
