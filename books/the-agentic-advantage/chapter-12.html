import anthropic

client = anthropic.Anthropic()

def run_agent_with_cache(system_context: str, user_message: str) -> str:
    """Run agent with prompt caching on static system context."""
    
    response = client.messages.create(
        model="claude-haiku-3-5",
        max_tokens=1024,
        system=[
            {
                "type": "text",
                "text": system_context,
                "cache_control": {"type": "ephemeral"}  # Cache this block
            }
        ],
        messages=[
            {"role": "user", "content": user_message}
        ]
    )
    
    return response.content[0].text

SNIPPET 7Prevention is structural. Every agent loop gets a hard maximum iteration count. Every agent run gets a timeout. Every agent that calls the A

python
import time

def run_agent_loop(agent_func, max_iterations: int = 100, timeout_seconds: int = 300):
    """Agent loop wrapper with hard limits."""
    
    start_time = time.time()
    iterations = 0
    
    while True:
        # Hard iteration cap
        if iterations >= max_iterations:
            send_alert(f"Agent hit max iterations ({max_iterations}). Halting.")
            break
            
        # Hard timeout
        if time.time() - start_time > timeout_seconds:
            send_alert(f"Agent exceeded timeout ({timeout_seconds}s). Halting.")
            break
        
        result = agent_func()
        iterations += 1
        
        if result.get('done'):
            break
    
    return iterations

SNIPPET 8Structure validation: For agents that produce structured output (JSON, CSV, formatted reports), add a validation step:

python
import json

def validate_agent_output(output: str, required_fields: list) -> bool:
    """Validate agent output has expected structure."""
    try:
        data = json.loads(output)
        
        for field in required_fields:
            if field not in data:
                send_alert(f"Agent output missing required field: {field}")
                return False
                
        return True
        
    except json.JSONDecodeError:
        send_alert("Agent output failed JSON parsing; possible format drift")
        return False

SNIPPET 9The most important fix is atomic writes. Never write directly to the target file:

python
import json
import os
import tempfile

def save_json_atomic(filepath: str, data: dict) -> None:
    """Write JSON atomically: either the full write succeeds or nothing changes."""
    
    dir_path = os.path.dirname(filepath)
    
    # Write to temp file in same directory (same filesystem)
    with tempfile.NamedTemporaryFile(
        mode='w',
        dir=dir_path,
        suffix='.tmp',
        delete=False
    ) as tmp_file:
        json.dump(data, tmp_file, indent=2)
        tmp_path = tmp_file.name
    
    # Atomic rename: on same filesystem, this is guaranteed atomic
    os.replace(tmp_path, filepath)

SNIPPET 10For shared files with concurrent access, add a file lock:

python
import fcntl  # Unix/Linux; on Windows use msvcrt.locking or a lock file pattern

def write_with_lock(filepath: str, data: dict) -> None:
    """Write JSON with file lock to prevent race conditions."""
    lock_path = filepath + '.lock'
    
    with open(lock_path, 'w') as lock_file:
        fcntl.flock(lock_file, fcntl.LOCK_EX)
        try:
            save_json_atomic(filepath, data)
        finally:
            fcntl.flock(lock_file, fcntl.LOCK_UN)
