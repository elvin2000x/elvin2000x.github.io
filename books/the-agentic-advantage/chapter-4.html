SNIPPET 4The Agent

python
import anthropic
import json
import os
from datetime import datetime

def run_briefing():
    with open("briefing_config.json") as f:
        config = json.load(f)

    client = anthropic.Anthropic()

    topics_str = ", ".join(config["topics"])
    query = f"Latest news and developments in {config['domain']}: {topics_str}. Focus on the last 24 hours."

    response = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1024,
        tools=[{
            "type": "web_search_20250305",
            "name": "web_search",
            "max_uses": 3
        }],
        messages=[{
            "role": "user",
            "content": f"""Search for recent developments in this domain and produce a morning briefing.

Domain: {config['domain']}
Topics to cover: {topics_str}

Return EXACTLY 5 bullet points. Each bullet: one sentence, specific and actionable.
Format: • [Topic]: [Finding]. Source: [publication name].
No intro, no conclusion, no filler. Just the 5 bullets."""
        }]
    )

    briefing_text = ""
    for block in response.content:
        if hasattr(block, "text"):
            briefing_text = block.text
            break

    today = datetime.now().strftime("%Y-%m-%d")
    filename = f"{config['output_folder']}briefing_{today}.md"
    os.makedirs(config["output_folder"], exist_ok=True)

    with open(filename, "w") as f:
        f.write(f"# Morning Briefing: {today}\n\n")
        f.write(briefing_text)
        f.write(f"\n\n---\nGenerated: {datetime.now().strftime('%H:%M')}\n")

    print(f"Briefing written to {filename}")

    log_run(today, len(briefing_text), response.usage.input_tokens + response.usage.output_tokens)

def log_run(date, chars, tokens):
    log_entry = {
        "date": date,
        "timestamp": datetime.now().isoformat(),
        "output_chars": chars,
        "tokens_used": tokens,
        "status": "success"
    }
    log_file = "briefing_log.json"
    log = []
    if os.path.exists(log_file):
        with open(log_file) as f:
            log = json.load(f)
    log.append(log_entry)
    with open(log_file, "w") as f:
        json.dump(log, f, indent=2)

if __name__ == "__main__":
    run_briefing()

SNIPPET 6The fix is the atomic write pattern:

python
import tempfile
import os

def save_state(state: dict, state_file: str):
    dir_name = os.path.dirname(state_file)
    with tempfile.NamedTemporaryFile(
        mode='w',
        dir=dir_name,
        delete=False,
        suffix='.tmp'
    ) as tmp:
        json.dump(state, tmp, indent=2)
        tmp_path = tmp.name
    os.replace(tmp_path, state_file)

SNIPPET 7The fix is structural: make reading memory the first thing the agent does, before any API calls, before any work. Not in the middle of the f

def run_agent():
    state = load_state()          # First
    context = build_context(state)  # Second

    new_state = update_state(state, results)  # Fourth
    save_state(new_state)         # Fifth (atomic)
